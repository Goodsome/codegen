@dataclass
class BaseTranslator:

    def _translate_attribute(
        self, attribute: Attribute, in_pydantic_model: bool
    ) -> ParameterSpec:
        return ParameterSpec.create(
            name=attribute.name,
            annotation=attribute.type,
            optional=attribute.optional,
            in_pydantic_model=in_pydantic_model,
        )

    def _translate_attributes(
        self, attributes: List[Attribute], in_pydantic_model: bool = False
    ) -> List[ParameterSpec]:
        return [self._translate_attribute(a, in_pydantic_model) for a in attributes]

    def _translate_method(
        self, method_spec: MethodSpec, is_abstract: bool = False
    ) -> FunctionSpec:
        parameter_specs = self._translate_attributes(
            method_spec.inputs, in_pydantic_model=False
        )
        if is_abstract:
            decorators = ["abstractmethod"]
        else:
            decorators = []
        return FunctionSpec(
            name=method_spec.name,
            decorators=decorators,
            parameters=parameter_specs,
            return_annotation=TypeAnnotationSpec.parse(method_spec.output.type),
            function_type=FunctionType.INSTANCE_METHOD,
        )

    def _translate_methods(
        self, method_specs: List[MethodSpec], is_abstract: bool = False
    ) -> List[FunctionSpec]:
        return [self._translate_method(m, is_abstract) for m in method_specs]

class AppTranslator(BaseTranslator):

    def translate_application(self, application: MetaApplication) -> PackageSpec:
        sub_packages = [
            self.translate_use_cases(application.use_cases),
        ]
        return PackageSpec.create(
            name="application",
            sub_packages=sub_packages,
        )

    def translate_use_case(self, use_case: MetaUseCase) -> List[ClassSpec]:
        result: List[ClassSpec] = []
        if use_case.kind == "command":
            command_name = f"{use_case.name}Command"
            command_class = ClassSpec(
                name=command_name,
                decorators=["dataclass(frozen=True)"],
                attributes=self._translate_attributes(use_case.command.attributes),
            )
            param = ParameterSpec(
                name="cmd",
                annotation=TypeAnnotationSpec.parse(command_name),
            )
            result.append(command_class)
        elif use_case.kind == "query":
            query_name = f"{use_case.name}Query"
            query_class = ClassSpec(
                name=query_name,
                decorators=["dataclass(frozen=True)"],
                attributes=self._translate_attributes(use_case.query.attributes),
            )
            param = ParameterSpec(
                name="query",
                annotation=TypeAnnotationSpec.parse(query_name),
            )
            result.append(query_class)
        else:
            raise ValueError(f"Unknown use case kind: {use_case.kind}")

        result_name = f"{use_case.name}Result"
        result_class = ClassSpec(
            name=f"{use_case.name}Result",
            decorators=["dataclass(frozen=True)"],
            attributes=self._translate_attributes(use_case.result.attributes),
        )
        result.append(result_class)

        use_case_class = ClassSpec(
            name=use_case.name,
            description=use_case.description,
            decorators=["dataclass"],
            attributes=self._translate_attributes(use_case.attributes),
            methods=[
                FunctionSpec(
                    name="execute",
                    parameters=[param],
                    return_annotation=TypeAnnotationSpec.parse(result_name),
                    function_type=FunctionType.INSTANCE_METHOD,
                )
            ],
        )
        result.append(use_case_class)
        return result

    def translate_use_cases(self, use_cases: List[MetaUseCase]) -> PackageSpec:
        modules: List[ModuleSpec] = []
        for use_case in use_cases:
            classes = self.translate_use_case(use_case)
            module_spec = ModuleSpec.create(
                name=use_case.name,
                classes=classes,
            )
            modules.append(module_spec)
        return PackageSpec.create(
            name="use_cases",
            modules=modules,
        )

@dataclass
class PythonSyntaxTranslator:
    """Bidirectional translator between Python source code and PackageSpec."""

    template_port: TemplatePort
    file_system_port: FileSystemPort

    def to_package_spec(self, package_path: Path) -> PackageSpec:
        """
        Reconstruct a PackageSpec from a dictionary of file paths and their contents.
        """

        if not self.file_system_port.is_directory(package_path):
            raise ValueError(f"Expected a directory, got {package_path}")
        package_name = package_path.stem
        modules: list[ModuleSpec] = []
        sub_packages: list[PackageSpec] = []
        for filepath in self.file_system_port.list_directory_flat(package_path):
            if self.file_system_port.is_file(filepath) and filepath.suffix == ".py":
                source_code = self.file_system_port.read_file(filepath)
                modules.append(ModuleSpec.parse_code(source_code, filepath.stem))
            elif self.file_system_port.is_directory(filepath):
                sub_packages.append(self.to_package_spec(filepath))

        return PackageSpec.create(
            name=package_name,
            modules=modules,
            sub_packages=sub_packages,
        )

    def to_code(
        self, module_spec: ModuleSpec, imports: Iterable[ImportFromSpec]
    ) -> str:
        context = {"module_spec": module_spec, "imports": imports}
        content = self.template_port.render("module.j2", context)
        return content

    def generate_source_tree(
        self, package_spec: PackageSpec, target_node: str | None
    ) -> dict[Path, str]:
        """
        核心方法：将 PackageSpec 转换为虚拟文件树。
        返回格式: { Path('relative/path/to/file.py'): 'source code content' }
        """
        # 1. 预先构建依赖解析器
        resolver = DependencyResolver.build_from_package_spec(package_spec)

        # 2. 结果容器
        virtual_files: dict[Path, str] = {}

        # 3. 递归生成
        self._collect_files_recursively(
            current_spec=package_spec,
            resolver=resolver,
            output_dict=virtual_files,
            current_path=Path(package_spec.name),  # 根目录名
            target_node=target_node,
        )

        return virtual_files

    def _collect_files_recursively(
        self,
        current_spec: PackageSpec,
        resolver: DependencyResolver,
        output_dict: dict[Path, str],
        current_path: Path,
        target_node: str | None,
    ) -> None:
        # 处理当前包下的 Modules
        for module in current_spec.modules:
            # 过滤逻辑下沉到这里
            if (
                target_node
                and target_node != module.name
                and not module.is_init_module()
            ):
                continue

            # 解析导入依赖
            imports = resolver.resolve_module(module)

            # 生成代码 (调用单文件生成逻辑)
            source_code = self.to_code(module, imports)

            # 记录文件路径和内容
            file_path = current_path / module.filename
            output_dict[file_path] = source_code

        # 处理子包 (递归)
        for subpackage in current_spec.sub_packages:
            self._collect_files_recursively(
                current_spec=subpackage,
                resolver=resolver,
                output_dict=output_dict,
                current_path=current_path / subpackage.name,
                target_node=target_node,
            )

## 1) 转化逻辑重、还带业务逻辑：为什么放在 orchestration 层会别扭？

**Orchestration（编排层）适合做流程调度，不适合承载复杂的规则与结构变换**。一旦把 mapping 放进去：

- 编排层开始“懂”两个上下文的模型细节（强耦合）
- 转化规则会膨胀，编排层变成“上帝对象”
- 未来你想复用转化（比如 CLI/HTTP/定时任务都要生成）会很痛

所以你把它移到 `python_gen/application/translators` 是一个合理方向：因为它更像“应用层的用例支撑组件”。

---

## 2) 需要在 MetaApplication 下增加 Translator 模型吗？通常**不合适**

你的担心点是对的：为了让 `AppTranslator.translate_application(application: MetaApplication)` 成立，你可能会想在 `MetaApplication` 里增加类似 `translators: ...` 或 `translator: ...`。

- `MetaApplication`（属于 `domain_definition` 上下文的领域模型）应该只表达“蓝图规格/业务概念”
- `Translator` 是“把 A 模型变成 B 模型”的**技术性策略/实现细节**
- 一旦你把 Translator 放进 `MetaApplication`，就会出现反向依赖：  
  **领域模型依赖应用层/其它上下文能力**（这是强烈的坏味道）

**结论：一般不应该把 Translator 作为领域模型的一部分。**  
（除非你真的在领域中有“可配置的生成策略”这一业务概念，并且它是纯数据描述，比如 `generation_profile: str` 这种 *配置值对象*，而不是 Translator 这种行为对象。）

def to_package_spec(self, package_path: Path) -> PackageSpec:
        """
        Reconstruct a PackageSpec from a dictionary of file paths and their contents.
        """

        if not self.file_system_port.is_directory(package_path):
            raise ValueError(f"Expected a directory, got {package_path}")
        package_name = package_path.stem
        modules: list[ModuleSpec] = []
        sub_packages: list[PackageSpec] = []
        for filepath in self.file_system_port.list_directory_flat(package_path):
            if self.file_system_port.is_file(filepath) and filepath.suffix == ".py":
                source_code = self.file_system_port.read_file(filepath)
                modules.append(ModuleSpec.parse_code(source_code, filepath.stem))
            elif self.file_system_port.is_directory(filepath):
                sub_packages.append(self.to_package_spec(filepath))

        return PackageSpec.create(
            name=package_name,
            modules=modules,

class PackageSpec(ValueObject):
    """Represents a Python package."""

    name: str
    modules: list[ModuleSpec] = Field(default_factory=list)
    sub_packages: list["PackageSpec"] = Field(default_factory=list)

    @classmethod
    def create(
        cls,
        name: str,
        modules: list[ModuleSpec] | None = None,
        sub_packages: list["PackageSpec"] | None = None,
    ):
        if modules is None:
            modules = []
        if sub_packages is None:
            sub_packages = []
        has_init_module = any(mod.is_init_module() for mod in modules)
        if not has_init_module:
            modules.append(ModuleSpec.get_init_module())
        return cls(
            name=name,
            modules=modules,
            sub_packages=sub_packages,
        )

    def get_global_registry(self, root_name: str = "") -> dict[str, str]:
        symbol_table: dict[str, str] = {}
        self._build_symbol_table(root_name, symbol_table)
        return symbol_table

    def _build_symbol_table(self, root_path: str, table: dict[str, str]):
        """递归扫描所有模块，记录每个类属于哪个绝对路径"""
        if root_path:
            current_path = f"{root_path}.{self.name}"
        else:
            current_path = self.name

        for mod in self.modules:
            mod_name = Path(mod.filename).stem
            full_mod_path = f"{current_path}.{mod_name}"
            for cls in mod.classes:
                table[cls.name] = full_mod_path

        for pkg in self.sub_packages:
            pkg._build_symbol_table(current_path, table)

- **短期（硬编码可接受）**：放在 ACL 实现里，没问题

- **中期（可配置）**：把“规则”从实现里抽离成一个 `GenerationStyle/Policy`（配置对象 + 策略类）

class YamlBlueprintLoader(BlueprintLoaderPort):
    """Blueprint loader that reads from a YAML file."""

    def __init__(self, config: dict[str, Any]) -> None:
        project_root = config.get("project_root", ".")
        if isinstance(project_root, str):
            self.project_root = Path(project_root)
        elif isinstance(project_root, Path):
            self.project_root = project_root
        else:
            raise ValueError("Invalid project_root value")

    def load(self, source: str) -> Blueprint | None:
        yaml_path = self.project_root / source
        if not os.path.exists(yaml_path):
            return None

        try:
            with open(yaml_path, "r", encoding="utf-8") as f:
                data = cast(dict[str, object], yaml.safe_load(f))

from typing import Iterable
from codegen.python_gen.domain.value_objects.package_spec import PackageSpec
from codegen.python_gen.domain.value_objects.import_from_spec import ImportFromSpec
from dataclasses import dataclass

from codegen.python_gen.domain.value_objects.module_spec import ModuleSpec

import logging

logger = logging.getLogger(__name__)


BUILTIN_TYPES = {
    "str",
    "int",
    "float",
    "bool",
    "None",
    "list",
    "dict",
}

GLOBAL_REGISTRY = {
    "Field": "pydantic",
    "Path": "pathlib",
    "Any": "typing",
    "Dict": "typing",
    "Union": "typing",
    "Iterable": "typing",
    "dataclass": "dataclasses",
    "ValueObject": "codegen.shared.models",
    "ABC": "abc",
    "abstractmethod": "abc",
}

TEMPORARY_MAPPING = {"dataclass(frozen=True)": "dataclass"}


@dataclass
class DependencyResolver:

    global_registry: dict[str, str]

    @classmethod
    def build_from_package_spec(cls, package_spec: PackageSpec) -> "DependencyResolver":
        global_registry = package_spec.get_global_registry()
        global_registry.update(GLOBAL_REGISTRY)
        return cls(global_registry=global_registry)

    def resolve_module(self, module_spec: ModuleSpec) -> Iterable[ImportFromSpec]:
        required_types = module_spec.get_required_types()
        import_spec_bags: dict[str, ImportFromSpec] = {}
        for rt in required_types:
            if module_spec.has_class_or_function(rt):
                continue
            if rt in BUILTIN_TYPES:
                continue
            rt = TEMPORARY_MAPPING.get(rt, rt)
            if rt not in self.global_registry:
                logger.warning(f"Could not find type {rt} in global registry")
                continue
            module_path = self.global_registry[rt]
            if module_path not in import_spec_bags:
                import_spec_bags[module_path] = ImportFromSpec(
                    module=module_path, names=[]
                )
            import_spec_bags[module_path].add_name(rt)

        return import_spec_bags.values()

    TypeAnnotationSpec,
)
from codegen.python_gen.domain.value_objects.function_spec import (
    FunctionSpec,
    FunctionType,
)
from .base import BaseTranslator


class AppTranslator(BaseTranslator):

    def translate_application(self, application: MetaApplication) -> PackageSpec:
        sub_packages = [
            self.translate_use_cases(application.use_cases),
        ]
        return PackageSpec.create(
            name="application",
            sub_packages=sub_packages,
        )

    def translate_use_case(self, use_case: MetaUseCase) -> List[ClassSpec]:
        result: List[ClassSpec] = []
        if use_case.kind == "command":
            command_name = f"{use_case.name}Command"
            command_class = ClassSpec(
                name=command_name,
                decorators=["dataclass(frozen=True)"],
                attributes=self._translate_attributes(use_case.command.attributes),
            )
            param = ParameterSpec(
                name="cmd",
                annotation=TypeAnnotationSpec.parse(command_name),
            )
            result.append(command_class)
        elif use_case.kind == "query":
            query_name = f"{use_case.name}Query"
            query_class = ClassSpec(
                name=query_name,
                decorators=["dataclass(frozen=True)"],
                attributes=self._translate_attributes(use_case.query.attributes),
            )
            param = ParameterSpec(
                name="query",
                annotation=TypeAnnotationSpec.parse(query_name),
            )
            result.append(query_class)
        else:
            raise ValueError(f"Unknown use case kind: {use_case.kind}")

        result_name = f"{use_case.name}Result"
        result_class = ClassSpec(
            name=f"{use_case.name}Result",
            decorators=["dataclass(frozen=True)"],
            attributes=self._translate_attributes(use_case.result.attributes),
        )
        result.append(result_class)

        use_case_class = ClassSpec(
            name=use_case.name,
            description=use_case.description,
            decorators=["dataclass"],
            attributes=self._translate_attributes(use_case.attributes),
            methods=[
                FunctionSpec(
                    name="execute",
                    parameters=[param],
                    return_annotation=TypeAnnotationSpec.parse(result_name),
                    function_type=FunctionType.INSTANCE_METHOD,
                )
            ],
        )
        result.append(use_case_class)
        return result

    def translate_use_cases(self, use_cases: List[MetaUseCase]) -> PackageSpec:
        modules: List[ModuleSpec] = []
        for use_case in use_cases:
            classes = self.translate_use_case(use_case)
            module_spec = ModuleSpec.create(
                name=use_case.name,
                classes=classes,
            )
            modules.append(module_spec)
        return PackageSpec.create(
            name="use_cases",
            modules=modules,
        )

Assistant: 
你这个目标（A：让 **yaml 与代码在整个开发周期保持一致**，并且反向只需要“约定级”）非常清晰，也非常适合用 **“强约定 + 受控可逆（round-trip）”** 的方式来做。

下面我按你关心的点给出一个可落地的架构方案：**Port 放哪、ACL 怎么放、如何保证可逆、以及如何支持“更新现有代码”而不是一把覆盖。**

---

## 1) 结论先说：Port 放 `python_gen/domain/ports`，ACL 放 `python_gen/infrastructure/acl/` ——完全可以

你现在的需求本质是：

- `python_gen` 需要把 `domain_definition.Blueprint` 映射为 `python_gen.PackageSpec`（正向）
- `python_gen` 也需要把 `python_gen.PackageSpec`（来自解析代码）映射回 `domain_definition.Blueprint`（反向）
- 这两个映射都属于跨上下文（ACL）

所以一个推荐的目录形态是：

```text
codegen/python_gen/
  domain/
    ports/
      blueprint_mapper.py        # 声明“可逆映射”能力（接口）
  infrastructure/
    acl/
      blueprint_mapper_v1.py     # 约定级实现（依赖 domain_definition 的模型）
```

> 关键点：**Port 属于 python_gen（调用方/依赖方）**；ACL 实现属于 python_gen 的 infrastructure（因为它跨上下文、且会经常变）。

---

## 2) 你说的“业务逻辑”（VO 基类、dataclass 装饰器）应当归类为“生成策略”，放在 ACL 是合理的，但要集中管理

这些规则不是 `domain_definition` 的领域事实，而是“Python 代码骨架风格/框架选型”的策略：

- VO 继承 `ValueObject(BaseModel)`
- command/query/result 用 `dataclass(frozen=True)` 等

**短期硬编码没问题**，但建议你不要让它散落在多个 translator 里，而是抽成一个“策略对象”，否则未来你做配置化会很痛。

推荐做法（概念示意）：

```python
class PythonDddStylePolicy:
    value_object_base = "ValueObject"
    command_decorator = "dataclass(frozen=True)"
    query_decorator = "dataclass(frozen=True)"
    result_decorator = "dataclass(frozen=True)"
```

ACL 映射时只引用 policy，不直接写死字符串到每个 translator 方法里。  
这样以后你把 policy 参数化（从 yaml 或 profile 来）也很自然。

---

## 3) 你要的 A（yaml <-> code 一致）如果只做“约定级”，**仍然建议加入“机器可读标记”，否则可逆性很难保证**

你说你不想“解析任意代码”，只想维护“约定”。这没问题——但要实现 **可靠的 round-trip**，至少要做到：

- 从代码里**无歧义地识别**：哪个 bounded context、哪个 layer、哪个 class 对应 blueprint 里的哪个元素
- 识别失败时，能明确报错/提示“这段代码不在约定内”

仅靠目录/命名约定（`domain/value_objects/*.py`、`*Command`）是可以做到“约定级”的，但会遇到几个不可避免的歧义：

- 类名改了、文件名改了：是“重命名”还是“删了又新增”？
- 同名类在不同模块出现：怎么回推？
- 你未来增加其他语言/布局：约定会变，回推规则要跟着变

### 推荐最小标记方案（强烈建议）
在每个生成的模块（或每个生成的类）里加入一个非常轻量的机器可读标记，例如模块级常量：

```python
__codegen__ = {
    "context": "payments",
    "layer": "domain.value_objects",
    "kind": "value_object",
    "name": "Money",
    "version": 1,
}
```

它不会让你“解析任意代码”，反而是让你**只解析你自己生成/受控的代码**：  
- 找不到 `__codegen__` 的模块：直接认为“不受控/不参与 round-trip”
- 找到了就按标记精确回推 blueprint 节点

这会让 A 的成功率从“八成靠运气”变成“几乎确定”。

> 如果你非常不想污染业务代码，也可以把标记放到 docstring 或固定格式注释块里，本质一样。

---

## 4) 反向转化（PackageSpec -> Blueprint）推荐实现为“约定解析器 + 容错策略”，并且**只对受控区域生效**

你说得很好：你不想做通用逆向工程，你只想维护约定。

那反向的边界应该是：

- **仅扫描** `codegen/<context>/domain|application|infrastructure` 这类受控路径
- **仅识别** 有 `__codegen__`（或标记块）的模块/类
- 解析失败就报“非约定/非受控”，而不是猜

这样反向逻辑会非常可控，也不会变成“解析任意项目结构”的大坑。

---

## 5) 更关键的一点：你真正要的是“同步”，不仅是“反向生成 yaml”

为了让“yaml 改动正确更新到现有代码（可能需要人工调整）”，你最终需要的其实是一个三段式能力：

1. **代码 -> PackageSpec**（你已有）
2. **PackageSpec -> Blueprint（约定级）**（你将要做）
3. **Blueprint(新) + Blueprint(由代码回推的旧) -> 生成变更计划（plan）**（建议补上）

第 3 步非常关键，它决定你是：

## 1) 同步更新 ≠ 仅“不覆盖 function body”这么简单，但它是核心原则

## 1) 受控模块标记规范：`__codegen__`

### 1.1 必备字段（推荐最小集合）
```python
__codegen__ = {
    "version": 1,
    "managed": True,

    "context": "<bounded_context_name>",      # 例如 "python_gen"
    "layer": "<domain|application|infrastructure>",

    "kind": "<aggregate|value_object|service|port|use_case|adapter|package_init>",
    "name": "<logical_name>",                 # 例如 "Money" / "LoadBlueprint" / "YamlBlueprintLoader"
}
```

### 1.2 可选字段（强烈建议，用于稳健回推与诊断）
- `qualified_name`: `"domain.value_objects.Money"`（用于调试/校验）
- `source`: `"blueprint"`（标记来源）
- `layout`: `"ddd_standard"`（与 Blueprint.layout 对齐）
- `id`: 稳定标识（未来支持重命名迁移时用；回推阶段可不必）

### 1.3 解析方式
- 解析器从模块源码 AST 提取 `__codegen__` 的 dict literal。
- 只接受**静态字面量**（字符串/数字/布尔/None/列表/字典），不执行代码。
- 若 `__codegen__` 不是字面量 dict：判为非受控（或报错）。

## 2) 目录约定规范（DDD Standard）

这是“约定级”回推的目录规则。假设 root package 名为 `codegen`（或你解析出的根包名）。

### 2.1 路径结构（强约定）
```text
<root_pkg>/<context_name>/
  domain/
    aggregates/
    value_objects/
    services/
    ports/
  application/
    use_cases/
  infrastructure/
    adapters/
```

### 2.2 目录到 Blueprint 节点的映射
- `<context_name>` -> `BoundedContext.name`
- `domain/aggregates` -> `MetaDomain.aggregates`
- `domain/value_objects` -> `MetaDomain.value_objects`
- `domain/services` -> `MetaDomain.services`
- `domain/ports` -> `MetaDomain.ports`
- `application/use_cases` -> `MetaApplication.use_cases`
- `infrastructure/adapters` -> `MetaInfrastructure.adapters`

### 2.3 目录规则的用途
- 当 `__codegen__` 缺少 `layer` 或 `kind` 时，可根据路径补全。
- 当 `__codegen__` 与路径冲突时，以 `__codegen__` 为准，但记录 warning：
  - 例如 `layer="domain"` 但文件在 `application/use_cases/`。

## 3) 命名约定规范（用于补全与校验）

命名仅作为兜底：补全缺失信息或检测一致性，不作为主判定。

### 3.1 Domain 层
- Aggregate：类名任意（建议 PascalCase），`__codegen__.kind == "aggregate"`
- ValueObject：类名任意，`__codegen__.kind == "value_object"`
- Service：类名任意，`__codegen__.kind == "service"`
- Port：类名任意，`__codegen__.kind == "port"`

**校验建议（可选）**
- VO 文件通常与类同名（`money.py` / `Money`），不强制但推荐。
- Port 通常为抽象类/协议；不强制，但可做 warning。

### 3.2 Application/use_cases 层（重点）
针对一个用例 `UseCaseName`，在同一模块里通常包含：
- `UseCaseName`（用例类）
- `UseCaseNameCommand` 或 `UseCaseNameQuery`
- `UseCaseNameResult`

命名推断（兜底）：
- 存在 `XCommand` => `MetaUseCase.kind="command"`
- 存在 `XQuery` => `MetaUseCase.kind="query"`
- 存在 `XResult` => result 结构来源

冲突处理：
- 同时存在 `XCommand` 与 `XQuery`：报错（或优先 `__codegen__` 指定 kind）
- 两者都没有：若 `__codegen__` 也无 kind -> 报错（无法约定级回推）

## 4) “标记/目录/命名”组合推断规则（可执行流程）

对每个模块（ModuleSpec 或源码文件）：

### Step 1：是否受控？
- 有 `__codegen__` 且 `managed=True` -> 继续
- 否则：跳过（不参与回推）

### Step 2：确定 BoundedContext
优先级：
1. `__codegen__["context"]`
2. 目录 `<root_pkg>/<context_name>/...` 推断
3. 否则报错（无法确定属于哪个 context）

### Step 3：确定 layer + kind
优先级：
1. `__codegen__["layer"]` + `__codegen__["kind"]`
2. 目录推断（见 2.2）
3. 命名推断（仅对 use_case kind 的 command/query 做兜底）
4. 否则报错

### Step 4：抽取模型对象
按 `(layer, kind)` 分派到 extractor：

- domain / aggregate -> MetaAggregate
- domain / value_object -> MetaValueObject
- domain / service -> MetaService
- domain / port -> MetaDomainPort
- application / use_case -> MetaUseCase
- infrastructure / adapter -> MetaInfrastructureAdapter

### Step 5：合并进 Blueprint
- 同一个 context 下按集合合并（按 `name` 唯一）。
- 若同名重复出现（两个受控模块都声称生成同一个 VO/UseCase 名称）：报错或保留一个并输出冲突列表（推荐报错，防止隐性错误）。

## 5) 从代码结构回推 Meta* 字段的规则（约定级）

> 你说“约定级即可”，那我们只回推“能从结构稳定得到的信息”。描述类字段可选，复杂语义不推断。

### 5.1 Attribute（字段）回推
- 从类的“属性定义”回推字段列表：
  - 优先：dataclass 字段 / pydantic model fields（如果你的 ModuleSpec 里能拿到）
  - 其次：`__init__` 参数（不推荐，太易受实现影响）

字段信息：
- `name`
- `type`（从注解字符串化；无法确定则用 `Any` 或 `str`，并记录 warning）
- `optional`（若注解是 `X | None` / `Optional[X]` 或有默认 None）

> 如果目前 PackageSpec/ModuleSpec 的解析能力不足以稳定拿到属性列表，规范上可以先定义为：**只回推 name/type（尽力），无法则空列表**。这也符合“约定级”的宽容策略。

### 5.2 MethodSpec（行为/操作）回推
对于：
- Aggregate.behaviors
- Service.operations
- Port.operations

约定级建议先回推：
- `name`
- `inputs`（参数名+注解）
- `output.type`（返回注解）
不回推：
- description（除非你从 docstring 提取）
- 复杂默认值/泛型细节（可后续增强）

### 5.3 UseCase 回推（关键）
从同模块内类集合回推：

- `MetaUseCase.name`：来自 `__codegen__["name"]` 或用例类名
- `MetaUseCase.kind`：
  - 优先 `__codegen__`
  - 否则看是否存在 `NameCommand` / `NameQuery`
- `MetaUseCase.command/query/result`：
  - 从对应 DTO 类字段回推 attributes
- `MetaUseCase.attributes`（用例自身依赖，如 ports）：
  - 可选回推：从用例类 attributes（dataclass fields）推断
  - 不可靠时可留空（约定级允许）

## 6) 冲突与错误处理规范（非常重要）

### 6.1 必须报错的情况（阻断回推）
- 同一个模块标记为受控，但缺少 `context` 且无法从路径推断
- `layer/kind` 无法确定
- use_case 同时出现 Command 与 Query 且标记未明确
- 同一 context 下同一 `kind` 出现两个同名实体（重复定义）

### 6.2 仅 warning 的情况（继续回推）
- `__codegen__` 与路径推断不一致
- 字段类型无法解析（降级为 `Any`/`str`）
- 缺少 description（留空）

### 6.3 回推输出应包含诊断报告
即使你先不实现，也建议规范里规定返回：
- `blueprint`
- `warnings: list[str]`
- `errors: list[str]`（若有则抛异常或返回失败）

- `Blueprint.layout`：优先从 `__codegen__.layout`；否则使用默认（如 `"ddd_standard"`）

- `bootstrap.bindings`：**不建议从代码回推**（除非你也把它标记化）；约定级可以：
