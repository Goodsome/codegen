### 2) 增加通用的 `render_to_file` 和 `ensure_package`
```python
def ensure_package(dir_path: Path):
    dir_path.mkdir(parents=True, exist_ok=True)
    init_file = dir_path / "__init__.py"
    if not init_file.exists():
        init_file.write_text("", encoding="utf-8")

def render_to_file(env: Environment, template_name: str, dest: Path, ctx: dict):
    dest.parent.mkdir(parents=True, exist_ok=True)
    tpl = env.get_template(template_name)
    dest.write_text(tpl.render(**ctx), encoding="utf-8")
    print(f"Generated: {dest}")
```

### 3) 增加生成 Application 层
目标：为每个 bounded context 生成一个 use case `generate_<context>_blueprint.py`（示例），同时生成 `__init__.py`。

```python
def generate_application(target_pkg_root: Path, blueprint_name: str, contexts: List[BoundedContext], env: Environment):
    app_dir = target_pkg_root / "application"
    ensure_package(app_dir)

    use_cases_dir = app_dir / "use_cases"
    ensure_package(use_cases_dir)

    # 生成一个总入口 use case（可选）
    ctx = {
        "blueprint_name": blueprint_name,
        "generated_at": datetime.utcnow().isoformat(),
        "contexts": [c.name for c in contexts],
    }
    render_to_file(env, "application/bootstrap_generate_all.py.j2", use_cases_dir / "bootstrap_generate_all.py", ctx)

    # 每个 context 一个用例骨架：先做最小可运行版本
    for bc in contexts:
        ctx = {
            "blueprint_name": blueprint_name,
            "context_name": bc.name,
            "context_pascal": to_pascal(bc.name),
            "generated_at": datetime.utcnow().isoformat(),
            "aggregates": [a.name for a in bc.aggregates],
            "value_objects": [v.name for v in bc.value_objects],
        }
        render_to_file(
            env,
            "application/bootstrap_generate_context.py.j2",
            use_cases_dir / f"bootstrap_generate_{to_snake(bc.name)}.py",
            ctx
        )
```

### 4) 增加生成 Infrastructure 层（Adapters：jinja + os file system）
你项目里已经有 `src/codegen/infrastructure/adapters/jinja_adapter.py` 等，但 bootstrap 阶段不要依赖它们存在，所以直接生成一份“最小可用版”骨架。

```python
def generate_infrastructure(target_pkg_root: Path, env: Environment):
    infra_dir = target_pkg_root / "infrastructure"
    ensure_package(infra_dir)

    adapters_dir = infra_dir / "adapters"
    ensure_package(adapters_dir)

    ctx = {"generated_at": datetime.utcnow().isoformat()}

    render_to_file(env, "infrastructure/jinja_adapter.py.j2", adapters_dir / "jinja_adapter.py", ctx)
    render_to_file(env, "infrastructure/os_file_system.py.j2", adapters_dir / "os_file_system.py", ctx)
```

### 5) 增加生成 Interfaces 层（CLI）
生成一个最小 CLI（typer），至少提供一个命令 `bootstrap`，后续你再逐步替换成你现在已有的 `cli.py`（或直接模板生成那份）。

```python
def generate_interfaces(target_pkg_root: Path, env: Environment):
    interfaces_dir = target_pkg_root / "interfaces"
    ensure_package(interfaces_dir)

    ctx = {"generated_at": datetime.utcnow().isoformat()}
    render_to_file(env, "interfaces/cli.py.j2", interfaces_dir / "cli.py", ctx)
```

### 6) 在 `main()` 里组装 Environment，并调用三层生成
你现在 `template_dir = Path(__file__).parent / "codegen" / "templates"` 用于 domain 模板。我们继续复用它，但需要新增模板目录结构（下一节会给）。

在 `main()` 里，找到你创建 env 的地方（目前在 generate_aggregate/value_object 内部各创建一次 env），建议在 main 统一创建一个 env 并传下去，避免重复。

最小改动做法：在 `main()` 里新增 env，然后 domain 生成函数也改造一下接收 env（可选）。为了少改，我们只给 application/infra/interfaces 使用一个 env。

在 `main()` 开头加：

```python
    template_dir = root / "src" / "codegen" / "templates"
    env = Environment(loader=FileSystemLoader(template_dir))
```

然后在生成完 domain 后，追加：

```python
    # --- application / infrastructure / interfaces ---
    # 注意：target_pkg_root 是 src/codegen
    target_pkg_root = root / "src" / "codegen"

    # 把 contexts 也 parse 出来传进去
    parsed_contexts = [parse_context(c) for c in data.get("contexts", [])]

    generate_application(target_pkg_root, data["name"], parsed_contexts, env)
    generate_infrastructure(target_pkg_root, env)
    generate_interfaces(target_pkg_root, env)
```

> 你现在 domain 的 target_root 是 `root/src/codegen/domain`，而 app/infra/interfaces 是 `root/src/codegen/*`，所以这里用 `target_pkg_root = root/src/codegen`。

```jinja2
Generated at: {{ generated_at }}
```

```jinja2
from pathlib import Path
from jinja2 import Environment, FileSystemLoader


class JinjaTemplateAdapter:
    def __init__(self, template_dir: Path):
        self._env = Environment(loader=FileSystemLoader(str(template_dir)))

    def render(self, template_name: str, ctx: dict) -> str:
        tpl = self._env.get_template(template_name)
        return tpl.render(**ctx)
```
